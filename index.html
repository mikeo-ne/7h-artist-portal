        const ADMIN_PASSCODE = "7H2026"; // Change to your preferred passcode

        let isAdminAuthenticated = false;

        let submissions = [];

        let currentSelectedId = null;


        window.onload = function() {

            lucide.createIcons();

        function accessManagerPortal() {

            if (isAdminAuthenticated) {

                switchView('manager');

                return;

            }


            const inputCode = prompt("🔒 7H MUSIC GROUP ADMIN ACCESS\nEnter manager access key:");


            if (inputCode === ADMIN_PASSCODE) {

                isAdminAuthenticated = true;

                switchView('manager');

            } else if (inputCode !== null) {

                alert("❌ Access Denied: Invalid 7H Admin Key.");

            }

        }


        function switchView(view) {

            const artistView = document.getElementById('view-artist');

            const managerView = document.getElementById('view-manager');


            if (view === 'artist') {

                artistView.classList.remove('hidden');

                managerView.classList.add('hidden');

            } else {

                artistView.classList.add('hidden');

                managerView.classList.remove('hidden');

                renderSubmissionsList();

            }

        }


        function loadSubmissions() {

            const saved = localStorage.getItem('7h_artist_submissions');

            if (saved) {

                try { submissions = JSON.parse(saved); } catch (e) { submissions = []; }

            }

        }


        function saveToStorage() {

            localStorage.setItem('7h_artist_submissions', JSON.stringify(submissions));

        }


        function handleArtistSubmit(event) {

            event.preventDefault();

            const newSubmission = {

                id: 'artist-' + Date.now(),

                artistName: document.getElementById('artistName').value,

                legalName: document.getElementById('legalName').value,

                genre: document.getElementById('genre').value,

                location: document.getElementById('location').value,

                email: document.getElementById('email').value,

                phone: document.getElementById('phone').value,

                status: 'PENDING',

                scores: { m1: 8, m2: 7 }

            };


            submissions.unshift(newSubmission);

            saveToStorage();

            alert("Application submitted successfully to 7H Management!");

            document.getElementById('artist-onboarding-form').reset();

        }


        function renderSubmissionsList() {

            const container = document.getElementById('submissions-list-container');

            container.innerHTML = '';

            submissions.forEach(artist => {

                const card = document.createElement('div');

                card.className = "p-4 rounded-2xl border border-darkborder bg-darkcard text-xs text-white space-y-1";

                card.innerHTML = `<strong>${artist.artistName}</strong> (${artist.genre}) - <span class="text-purple-400">${artist.status}</span>`;

                container.appendChild(card);

            });

        }


        function updateScorecard() {

            const m1 = document.getElementById('m-score-1').value;

            const m2 = document.getElementById('m-score-2').value;

            document.getElementById('score-val-1').innerText = `${m1}/10`;

            document.getElementById('score-val-2').innerText = `${m2}/10`;

        }
